(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (nth m 0))
;;=> "123,456,789"
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (nth m 2))
;;=> "456"
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (nth m 3))
;;=> "789"
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (mapv #(nth m %) (range 4)))
;;=> ["123,456,789" "123" "456" "789"]
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (nth m 10 :default))
;;=> :default
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (try (nth m 10) (catch Exception e :threw)))
;;=> :threw
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (try (nth m -1) (catch Exception e :threw)))
;;=> :threw
(let [m (re-matcher #"(\d+),(\d+),(\d+)" "123,456,789")] (re-find m) (try (nth m -1 :default) (catch Exception e :threw)))
;;=> :default
(let [m (re-matcher #"(a)(b)?" "a")] (re-find m) (nth m 2))
;;=> nil
(let [m (re-matcher #"x" "abc")] (try (nth m 0) (catch Exception e :threw)))
;;=> :threw
