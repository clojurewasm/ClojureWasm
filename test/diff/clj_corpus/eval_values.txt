;;=> true

;;=> 42
(eval 5)
;;=> 5
(eval "s")
;;=> "s"
(eval :k)
;;=> :k
(eval \a)
;;=> \a
(eval nil)
;;=> nil
(eval true)
;;=> true
(eval 5N)
;;=> 5N
(eval 1/2)
;;=> 1/2
(eval #"a+")
;;=> #"a+"
(eval '(+ 1 2))
;;=> 3

;;=> 3
(eval [1 2 3])
;;=> [1 2 3]

;;=> {:a 1}

;;=> #{1}

;;=> ()

;;=> x

;;=> true

;;=> :threw
(try (eval (seq [:a])) (catch Exception e :threw))
;;=> :threw
