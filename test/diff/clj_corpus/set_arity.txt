(clojure.set/union)
;;=> #{}
(sort (clojure.set/intersection #{1 2 3}))
;;=> (1 2 3)
(sort (clojure.set/intersection #{1 2 3} #{2 3 4}))
;;=> (2 3)
(sort (clojure.set/intersection #{1 2 3} #{2 3 4} #{3 4 5}))
;;=> (3)
(sort (clojure.set/difference #{1 2 3}))
;;=> (1 2 3)
(sort (clojure.set/difference #{1 2 3} #{2}))
;;=> (1 3)
(sort (clojure.set/difference #{1 2 3 4} #{2} #{4}))
;;=> (1 3)
(clojure.set/subset? #{1} #{1 2})
;;=> true
(clojure.set/subset? #{1 5} #{1 2})
;;=> false
(clojure.set/superset? #{1 2} #{1})
;;=> true
(clojure.set/superset? #{1} #{1 2})
;;=> false
