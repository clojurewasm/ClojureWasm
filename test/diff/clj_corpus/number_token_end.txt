(read-string "2r")
;;=> <clj-error> NumberFormatException
(read-string "0x1g")

(read-string "+12x")
;;=> <clj-error> NumberFormatException
(read-string "1.5Mx")

(read-string "09")
;;=> <clj-error> NumberFormatException
(read-string "[1#{}]")

(read-string "[1'a]")
;;=> [1 (quote a)]
(read-string "[12(x)]")
;;=> [12 (x)]
(read-string "[1%]")
;;=> [1 %]
(read-string "0xFFN")
